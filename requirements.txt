# Python dependencies for How To Be Human
# New machine setup: run setup.bat (installs Python via winget, then these deps)
# Manual install (Python already present): py -m pip install -r requirements.txt

pygame-ce        # core engine (do NOT also install plain 'pygame' - they conflict)
PySide6          # editor UI (editor/main.py)
Pillow           # image loading for editor asset import (editor/asset_import.py, panels/details.py)
jsonschema       # validates data/ JSON against data/schemas/
opencv-python    # cutscene playback (engine/video.py) — OPTIONAL: absent = cutscene gracefully skips

# Only needed if you want to build the .exe with build.bat:
pyinstaller
